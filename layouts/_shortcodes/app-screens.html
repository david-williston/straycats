{{- /* Every app screenshot in the page bundle, newest first, as phone-shaped thumbnails linking to the full image, each with its date and caption underneath. Picks up files named NNNN-YYYY-MM-DD--.jpg, so adding a screenshot needs only the file. Captions come from data/app_screens.yaml (number -> en/es); without an entry the filename description is used (see the app-shot-caption partial). Commit details (subject, first paragraph of the body, GitHub link) come from data/app_commits.json (generated by scripts/git-history.mjs); a missing commit is skipped. Usage: {{< app-screens >}} */ -}} {{- $shots := slice }} {{- range .Page.Resources.ByType "image" }} {{- if findRE `^\d{4}-\d{4}-\d{2}-\d{2}-.+\.jpg$` (path.Base .Name) }}{{ $shots = $shots | append . }}{{ end }} {{- end }}
{{- range $i, $img := sort $shots "Name" "desc" }} {{- $name := path.Base $img.Name }} {{- $caption := partial "app-shot-caption.html" $img }} {{- $date := time.AsTime (substr $name 5 10) | time.Format ":date_long" }} {{- $big := $img.Resize (printf "%dx webp q85" (math.Min $img.Width 2400 | int)) }} {{- $s1 := $img.Resize "400x webp q80" }} {{- $s2 := $img.Resize "700x webp q80" }} {{- $commit := dict }} {{- $hash := index (split $name "-") 4 }} {{- with index (hugo.Data.app_commits | default dict) $hash }}{{ $commit = . }}{{ end }} {{- end }}
