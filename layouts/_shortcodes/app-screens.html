A slideshow (assets/js/slideshow.js) plays them, oldest first; the grid is only a fallback for browsers without JavaScript. Captions come from data/app_screens.yaml (number -> en/es); without an entry the filename description is used. Usage: {{< app-screens >}} */ -}} {{- $shots := slice }} {{- range .Page.Resources.ByType "image" }} {{- if findRE `^\d{4}-\d{4}-\d{2}-\d{2}-.+\.jpg$` (path.Base .Name) }}{{ $shots = $shots | append . }}{{ end }} {{- end }} {{- $lang := site.Language.Lang }} {{- $captions := hugo.Data.app_screens | default dict }} {{- /* Caption for a screenshot: data file entry in this language, else the humanized filename slug */}} {{- $asc := sort $shots "Name" "asc" }} {{- $total := len $asc }}
